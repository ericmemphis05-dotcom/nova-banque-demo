document.getElementById('transferForm').onsubmit = e => { e.preventDefault(); db = load(); const user = db.users.find(u => u.id === current?.id); if (!user) { toast('Session introuvable. Reconnectez-vous.'); logout(); return; } const email = document.getElementById('beneficiaryEmail').value.trim().toLowerCase(); const name = document.getElementById('beneficiaryName').value.trim(); const amount = Math.round(Number(document.getElementById('transferAmount').value) * 100) / 100; const reason = document.getElementById('transferReason').value.trim() || 'Virement'; if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) || !name) { toast('Vérifiez le nom et l’adresse e-mail du bénéficiaire.'); return; } if (!Number.isFinite(amount) || amount <= 0 || amount > 100000) { toast('Veuillez saisir un montant valide.'); return; } if (amount > Number(user.balance)) { toast('Solde démo insuffisant.'); return; } if (email === user.email) { toast('Vous ne pouvez pas vous désigner comme bénéficiaire.'); return; } const ref = 'NOVA-' + id(); const date = new Date().toISOString(); // Débit fictif de l'expéditeur user.balance = Math.round((Number(user.balance) - amount) * 100) / 100; db.transactions.push({ id: id(), userId: user.id, type: 'Virement fictif envoyé', detail: name + ' (' + email + ') – ' + reason, amount: -amount, date, reference: ref }); // Crédit du bénéficiaire uniquement s'il possède un compte local Nova const recipient = db.users.find(u => u.email === email); if (recipient) { recipient.balance = Math.round((Number(recipient.balance) + amount) * 100) / 100; db.transactions.push({ id: id(), userId: recipient.id, type: 'Virement fictif reçu', detail: 'De ' + user.name + ' – ' + reason, amount, date, reference: ref }); } // Journal de notification, même sans compte Nova db.emails.push({ id: id(), to: email, name, from: user.name, amount, reason, ref, date, delivered: false, status: 'notification non envoyée', recipientHasNovaAccount: Boolean(recipient) }); save(); current = user; e.target.reset(); render(); go('transfer'); toast('Simulation enregistrée. Réf. ' + ref + '. Aucun e-mail ni argent réel envoyé.'); };
